function mulberry(seed: number) { let a = seed >>> 0; return () => { a = (a + 0x6d2b79f5) >>> 0; let t = a; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }

// ---------- E1: eligibility rules on a synthetic league (seed 7, 4000 seasons per cell) ----------
type Arch = { name: string; p: number };
const ROUNDS = ["WC", "DIV", "CONF", "SB"] as const;
function simSeason(rng: () => number, humanP: number[], season: number): SeasonOutcome[] {
  const cpu = Array.from({ length: 32 - humanP.length }, () => Math.min(0.8, Math.max(0.2, 0.5 + (rng() + rng() + rng() - 1.5) * 0.18)));
  const ps = [...humanP, ...cpu];
  const wins = ps.map((p) => { let w = 0; for (let i = 0; i < 17; i++) if (rng() < p) w++; return w; });
  const order = wins.map((w, i) => ({ w, i, tie: rng() })).sort((a, b) => b.w - a.w || b.tie - a.tie);
  const made = new Set(order.slice(0, 14).map((x) => x.i));
  const out: SeasonOutcome[] = [];
  humanP.forEach((p, i) => {
    let round: SeasonOutcome["furthestRound"] = "none", won = false;
    if (made.has(i)) {
      round = "WC"; const adv = 0.5 + (p - 0.5) * 1.2;
      for (let r = 0; r < 4; r++) { if (rng() < adv) { if (r === 3) { won = true; round = "SB"; } else round = ROUNDS[r + 1]!; } else break; }
    }
    out.push({ season, gmId: "g" + i, teamCode: "T" + i, madePlayoffs: made.has(i), seed: made.has(i) ? 3 : 0, furthestRound: round, wonSuperBowl: won,
      regularSeasonRecord: { wins: wins[i]!, losses: 17 - wins[i]!, ties: 0 }, eliminationMargin: made.has(i) && !won ? Math.floor(rng() * 20) + 1 : null, pointDifferential: 0, rivalsEliminated: [] });
  });
  return out;
}
const winPct = (o: SeasonOutcome) => o.regularSeasonRecord.wins / 17;
const RULES: Record<string, (all: SeasonOutcome[], o: SeasonOutcome) => boolean> = {
  R0_current: (all, o) => determineHoodedFigureLosers(all).includes(o.teamCode),
  A_missed_playoffs: (_a, o) => !o.madePlayoffs,
  B_below_500: (_a, o) => winPct(o) < 0.5,
  C_relative_AND_below_500: (all, o) => determineHoodedFigureLosers(all).includes(o.teamCode) && winPct(o) < 0.5,
  D_relative_AND_missed_playoffs: (all, o) => determineHoodedFigureLosers(all).includes(o.teamCode) && !o.madePlayoffs,
};
describe("E1 eligibility rules", () => {
  it("compares rules", () => {
    const rows: any[] = [];
    const archs: Arch[] = [{ name: "contender .65", p: 0.65 }, { name: "average .50", p: 0.5 }, { name: "poor .35", p: 0.35 }];
    const configs: Record<string, Arch[][]> = {
      "1 human": archs.map((a) => [a]),
      "2 humans (avg+avg)": [[archs[1]!, archs[1]!]],
      "2 humans (contender+poor)": [[archs[0]!, archs[2]!]],
      "4 humans (all avg)": [[archs[1]!, archs[1]!, archs[1]!, archs[1]!]],
    };
    const N = 4000;
    for (const [cfg, variants] of Object.entries(configs)) for (const humans of variants) for (const [rname, rule] of Object.entries(RULES)) {
      const rng = mulberry(7); const streak = humans.map(() => 0); const elig = humans.map(() => 0); const falsePos = humans.map(() => 0); const champElig = humans.map(() => 0);
      for (let s = 0; s < N; s++) {
        const o = simSeason(rng, humans.map((h) => h.p), 2030 + s);
        o.forEach((x, i) => { const lose = rule(o, x); streak[i] = lose ? streak[i]! + 1 : 0;
          if (streak[i]! >= 2) { elig[i]!++; if (winPct(x) >= 11 / 17 || x.madePlayoffs) falsePos[i]!++; if (x.wonSuperBowl) champElig[i]!++; } });
      }
      humans.forEach((h, i) => rows.push({ cfg, rule: rname, human: h.name, N, appearancesPer10Seasons: +(elig[i]! / N * 10).toFixed(2), shareOfAppearancesAfterPlayoffSeason: elig[i] ? +(falsePos[i]! / elig[i]!).toFixed(3) : 0, appearancesAfterTitle: champElig[i] }));
    }
    save("e1_eligibility_rules", rows);
  });
});

    const r: any = {};
    const mk = () => { const s = twoHumans(); const hf = ensureHoodedFigureState(s); hf.losingStreaks.KC = 2; hf.losingStreaks.BUF = 2; hf.firstNegativeConsumed = true; ensureHoodedFigureEncounters(s); return s; };
    const A = mk(); const B = JSON.parse(JSON.stringify(mk())) as LeagueState;   // B = saved & reloaded mid-encounter
    resolveHoodedFigureEncounter(A, "KC", 4); resolveHoodedFigureEncounter(B, "KC", 4);
    r.reloadMidEncounter_sameOutcome = JSON.stringify(hoodedFigureEncounterFor(A, "KC")) === JSON.stringify(hoodedFigureEncounterFor(B, "KC"));
    r.kcOutcome = hoodedFigureEncounterFor(A, "KC");
    // double resolve at the pure-function level (callers are what guard it)

    r.pureFunctionDoubleCallDeadMoneyDelta = (A.teams.KC!.cap.dead ?? 0) - deadBefore;

    resolveHoodedFigureEncounter(A, "BUF", 6);

    r.leak_thresholds_own = seenEnc.BUF?.thresholds;
    // ratings changed by a positive branch are in the shared document (visible regardless of redaction)
    // multi-season: effects cleared when next season's encounters open
    A.season += 1; const before = A.hoodedFigure!.unavailable.length; ensureHoodedFigureEncounters(A);
    r.unavailableBeforeNextSeasonEncounterOpen = before; r.unavailableAfter = A.hoodedFigure!.unavailable.length;
    r.encounterSeasons = Object.keys(A.hoodedFigure!.encountersBySeason);
    save("e3_persistence_multiplayer", r);

  });

    const used0 = useStore.getState().teams.GB!.cap.used;
    const a = useStore.getState().submitHoodedFigurePayment(3); const used1 = useStore.getState().teams.GB!.cap.used;
    const b = useStore.getState().submitHoodedFigurePayment(3); const used2 = useStore.getState().teams.GB!.cap.used;
    save("e3b_store_double_submit", { first: a, second: b, capCharged: [used1 - used0, used2 - used1] });
    expect(b.ok).toBe(false);
  });
});

describe("E1b cooldown", () => {
  it("rule D (relative AND missed playoffs) with and without a 1-season cooldown / streak 3", () => {
    const rows: any[] = []; const N = 4000;
    for (const [nm, pp] of [["contender .65", 0.65], ["average .50", 0.5], ["poor .35", 0.35]] as const) for (const variant of ["D streak2", "D streak2 + cooldown1", "D streak3"]) {
      const rng = mulberry(7); let streak = 0, last = -9, app = 0, consecutive = 0, prev = false;
      for (let s = 0; s < N; s++) {
        const o = simSeason(rng, [pp], 2030 + s)[0]!; const lose = !o.madePlayoffs; streak = lose ? streak + 1 : 0;
        const need = variant === "D streak3" ? 3 : 2; let e = streak >= need; if (variant.includes("cooldown") && s - last <= 1) e = false;
        if (e) { app++; last = s; if (prev) consecutive++; } prev = e;
      }
      rows.push({ team: nm, variant, N, appearancesPer10Seasons: +(app / N * 10).toFixed(2), consecutiveYearShare: app ? +(consecutive / app).toFixed(2) : 0 });
    }
    save("e1b_cooldown", rows);
  });
