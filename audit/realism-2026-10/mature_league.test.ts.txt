// scratch, never committed: plays a CPU-only league through N seasons with the production stage machinery
import { mkdirSync, writeFileSync } from "node:fs";
import { describe, it } from "vitest";
import type { LeagueState } from "@/domain";
import { createLeague, DEFAULT_CONFIG } from "@/state/seed.ts";
import { onStageEntered, readyUpLocal, advanceStage } from "../src/phases.js";

const N = { starters: { QB: 1, RB: 1, WR: 3, TE: 1, OT: 2, OG: 2, C: 1, EDGE: 2, DT: 2, ILB: 2, CB: 3, S: 2 } as Record<string, number> };
describe("mature", () => { it("run", () => {
  const seed = Number(process.env.M_SEED ?? 1), talent = (process.env.M_TALENT ?? "realistic") as any, years = Number(process.env.M_YEARS ?? 2);
  const snapAt = new Set((process.env.M_SNAP ?? "1,3,5,10").split(",").map(Number));
  const OUT = process.env.AUDIT_OUT!; mkdirSync(OUT, { recursive: true });
  const s: LeagueState = createLeague(seed, { ...DEFAULT_CONFIG, humanGmCount: 1, talentImpact: talent });
  for (const g of s.gms) g.isHuman = false;
  s.stage = "fantasyDraft"; onStageEntered(s, "setup");
  const rostered = () => Object.values(s.players as Record<string, any>).filter((p) => p.nfl_team && p.nfl_team !== "FA" && !String(p.nfl_team).startsWith("PS:") && !p.free_agent && !p.retired);
  const quality = () => {
    const by: Record<string, any[]> = {};
    for (const p of rostered()) (by[p.nfl_team] ??= []).push(p);
    const rows = Object.values(by).map((ps) => {
      const v: number[] = []; let qb = 0;
      for (const [pos, n] of Object.entries(N.starters)) { const o = ps.filter((p) => p.position === pos).map((p) => p.overall).sort((a, b) => b - a).slice(0, n); v.push(...o); if (pos === "QB") qb = o[0] ?? 0; }
      return { m: v.reduce((a, b) => a + b, 0) / Math.max(1, v.length), qb };
    });
    const sd = (x: number[]) => { const m = x.reduce((a, b) => a + b, 0) / x.length; return Math.sqrt(x.reduce((a, b) => a + (b - m) ** 2, 0) / x.length); };
    return { teams: rows.length, starterSd: sd(rows.map((r) => r.m)), starterMean: rows.reduce((a, r) => a + r.m, 0) / rows.length, qbSd: sd(rows.map((r) => r.qb)), qbMean: rows.reduce((a, r) => a + r.qb, 0) / rows.length };
  };
  const out: any = { seed, talent, seasons: [], snapshots: {} };
  let idx = 0, lastKey = "", captured = new Set<number>(), snapped = new Set<number>(), guard = 0;
  while (idx <= years && guard++ < 6000) {
    const key = `${s.season}|${s.stage}|${s.week}`;
    if (key !== lastKey) {
      lastKey = key;
      if (s.stage === "preseason" && s.week === 1 && !snapped.has(s.season)) {
        snapped.add(s.season); idx++;
        const all = Object.values(s.players as Record<string, any>).filter((p) => !p.retired);
        const ros = rostered();
        const sup = { rostered: ros.length, fa: all.filter((p) => p.free_agent).length, meanAge: ros.reduce((n, p) => n + p.age, 0) / ros.length, n80: ros.filter((p) => p.overall >= 80).length, n70: ros.filter((p) => p.overall >= 70).length, rookies: ros.filter((p) => (p.years_pro ?? 99) <= 0).length, under24: ros.filter((p) => p.age <= 23).length, totalPlayers: all.length, qb80: ros.filter((p) => p.position === "QB" && p.overall >= 80).length };
        out.seasons.push({ idx, season: s.season, quality: quality(), supply: sup });
        if (snapAt.has(idx)) {
          const teams: Record<string, any[]> = {};
          for (const p of rostered()) (teams[p.nfl_team === "LAR" ? "LA" : p.nfl_team] ??= []).push(p);
          out.snapshots[idx] = teams;
        }
        if (idx > years) break;
      }
      if ((s.stage === "endOfSeasonConsolation" || s.stage === "offseasonHotSeat") && !captured.has(s.season)) {
        captured.add(s.season);
        const cur = out.seasons[out.seasons.length - 1];
        cur.games = s.games.filter((g: any) => g.phase === "REG" && g.played).map((g: any) => [g.homeTeam, g.awayTeam, g.homeScore, g.awayScore]);
        cur.po = s.games.filter((g: any) => g.phase !== "REG" && g.phase !== "PRE" && g.played).map((g: any) => [g.phase, g.homeTeam, g.awayTeam, g.homeScore, g.awayScore]);
        cur.champion = s.bracket?.champion ?? null;
      }
    }
    let moved = false;
    try { moved = readyUpLocal(s); } catch (e: any) { out.error = String(e.message).slice(0, 300); break; }
    if (!moved) { const r = advanceStage(s); if (!r.moved && !r.playedRound) { out.error = "STUCK " + key; break; } }
  }
  writeFileSync(`${OUT}/mature_${talent}_${seed}.json`, JSON.stringify(out));
}, 3000000); });
